cmake_minimum_required(VERSION 3.14)
project(pan_login_native_fixture LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(WRAPPER "" CACHE PATH "Flutter C++ wrapper")
set(DWW "" CACHE PATH "Composer SDK includes")
if(NOT EXISTS "${WRAPPER}/standard_codec.cc" OR NOT EXISTS "${DWW}/desktop_webview_window/webview_window_composer.h")
  message(FATAL_ERROR "Set WRAPPER and DWW to the installed Flutter wrapper and composer SDK include paths")
endif()
add_executable(pan_login_native_fixture pan_web_login_composer_test.cpp "${WRAPPER}/standard_codec.cc" "${WRAPPER}/core_implementations.cc")
target_include_directories(pan_login_native_fixture PRIVATE "${WRAPPER}/include" "${WRAPPER}/.." "${DWW}")
target_compile_definitions(pan_login_native_fixture PRIVATE NOMINMAX UNICODE _UNICODE)
target_compile_options(pan_login_native_fixture PRIVATE /W4 /WX /wd4100)
target_link_libraries(pan_login_native_fixture PRIVATE user32 gdi32 "${WRAPPER}/../flutter_windows.dll.lib")
add_custom_command(TARGET pan_login_native_fixture POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different "${WRAPPER}/../flutter_windows.dll" "$<TARGET_FILE_DIR:pan_login_native_fixture>")
